# LLM Bridge -- strategic directives for the native AI.
#
# This is the write half of the hybrid model. The LLM does not micromanage
# fronts; it sets intent, and the game's own AI executes it. The mechanism is
# deliberately boring: the harness sets a country flag or variable, an
# ai_strategy here keys on it, and the vanilla AI starts behaving accordingly.
#
# Each block below is one directive the harness can raise or drop. Weights are
# starting points -- tune them against your own game rather than trusting these.
#
# VERIFY: ai_strategy `type` names and their valid scopes change between
# versions. Diff this file against common/ai_strategy in your install before a
# run that matters; an unrecognised type is silently ignored by the game, which
# means a directive that looks applied and does nothing.

# Country-targeted directives (invade, protect, contain, ...) are generated,
# one block per (directive, target), into llm_bridge_targets.txt -- see
# src/hoi4_harness/modgen.py. This file keeps the untargeted postures.

llmb_defensive_posture = {
	enable = {
		has_country_flag = llmb_posture_defensive
	}
	abort = {
		NOT = { has_country_flag = llmb_posture_defensive }
	}
	# Hold the line and ask for more units on the front rather than pushing.
	ai_strategy = {
		type = front_unit_request
		tag = ROOT
		value = 50
	}
	ai_strategy = {
		type = garrison
		value = 30
	}
}

llmb_offensive_posture = {
	enable = {
		has_country_flag = llmb_posture_offensive
	}
	abort = {
		NOT = { has_country_flag = llmb_posture_offensive }
	}
	ai_strategy = {
		type = front_unit_request
		tag = ROOT
		value = 100
	}
}
